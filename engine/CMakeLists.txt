cmake_minimum_required(VERSION 3.20)
project(torment_nexus_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/llama.cpp llama)
add_executable(torment-engine main.cpp)
target_link_libraries(torment-engine PRIVATE llama llama-common)
target_include_directories(torment-engine PRIVATE ../vendor/llama.cpp/vendor)
target_compile_definitions(torment-engine PRIVATE TORMENT_ENGINE_REV="922be44aa6ac81b46f092716351cddff1c1733a7")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" TORMENT_WRAPPER_SHA256)
target_compile_definitions(torment-engine PRIVATE TORMENT_WRAPPER_SHA256="${TORMENT_WRAPPER_SHA256}")
set_target_properties(torment-engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
